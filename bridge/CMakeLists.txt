cmake_minimum_required(VERSION 3.22)
project(b210_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UHD_SOURCE "" CACHE PATH "UHD host directory")
set(UHD_BUILD "" CACHE PATH "UHD build directory")
add_executable(usrp_bridge server.cpp)
set_target_properties(usrp_bridge PROPERTIES PREFIX "lib" SUFFIX ".so")
target_include_directories(usrp_bridge PRIVATE "${UHD_SOURCE}/include" "${UHD_BUILD}/include" "${UHD_BUILD}/../../prefix/include")
target_link_libraries(usrp_bridge PRIVATE "${UHD_BUILD}/lib/libuhd.so" log)
target_link_options(usrp_bridge PRIVATE -Wl,-z,max-page-size=16384)
target_compile_options(usrp_bridge PRIVATE -Wall -Wextra -Werror=return-type)
target_compile_options(usrp_bridge PRIVATE "@${UHD_BUILD}/../boost-includes.rsp")
add_executable(queue_tests EXCLUDE_FROM_ALL queue_tests.cpp)
target_link_options(queue_tests PRIVATE -Wl,-z,max-page-size=16384)
